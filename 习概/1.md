A good approach is **BFS + shortest-path DAG + meet-in-the-middle**. The large \(C\le 10^{12}\) rules out DP by toll value.

### Key idea

Run BFS from junction \(1\) and reverse BFS from junction \(n\). Let \(D\) be the shortest number of roads from \(1\) to \(n\).

An edge \(u\to v\) belongs to some shortest route exactly when

$$
d_1[u]+1+d_n[v]=D.
$$

Keeping only these edges gives a layered DAG: every edge goes from layer \(i\) to layer \(i+1\).

Choose one layer as a meeting layer. For each vertex \(v\) there:

* enumerate toll sums of all shortest prefixes \(1\to v\);
* enumerate toll sums of all shortest suffixes \(v\to n\).

If their tolls are \(a\) and \(b\), then \(c_v\) was counted twice, so the complete route costs

$$
a+b-c_v.
$$

Thus we need

$$
a+b\le C+c_v.
$$

Sort both lists and count valid pairs with two pointers.

We choose the meeting layer that minimizes the number of prefix/suffix paths that must be enumerated. Processing meeting vertices one at a time also keeps memory much lower.

```cpp
#include <bits/stdc++.h>
using namespace std;

using u64 = uint64_t;

const int INF_DIST = 1e9;
const u64 INF_COST = (u64)4e18;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    u64 C;
    cin >> n >> m >> C;

    vector<u64> c(n + 1);
    for (int i = 1; i <= n; ++i)
        cin >> c[i];

    vector<vector<int>> g(n + 1), rg(n + 1);
    vector<pair<int, int>> edges;
    edges.reserve(m);

    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        rg[v].push_back(u);
        edges.push_back({u, v});
    }

    auto bfs = [&](int s, const vector<vector<int>>& graph) {
        vector<int> dist(n + 1, INF_DIST);
        queue<int> q;

        dist[s] = 0;
        q.push(s);

        while (!q.empty()) {
            int u = q.front();
            q.pop();

            for (int v : graph[u]) {
                if (dist[v] == INF_DIST) {
                    dist[v] = dist[u] + 1;
                    q.push(v);
                }
            }
        }

        return dist;
    };

    vector<int> d1 = bfs(1, g);

    if (d1[n] == INF_DIST) {
        cout << 0 << '\n';
        return 0;
    }

    vector<int> dn = bfs(n, rg);
    int D = d1[n];

    // Vertices belonging to shortest routes, grouped by BFS layer.
    vector<vector<int>> layer(D + 1);

    for (int v = 1; v <= n; ++v) {
        if (d1[v] != INF_DIST &&
            dn[v] != INF_DIST &&
            d1[v] + dn[v] == D) {
            layer[d1[v]].push_back(v);
        }
    }

    // Shortest-path DAG.
    vector<vector<int>> nxt(n + 1), prv(n + 1);

    for (auto [u, v] : edges) {
        if (d1[u] != INF_DIST &&
            dn[v] != INF_DIST &&
            d1[u] + 1 + dn[v] == D) {
            nxt[u].push_back(v);
            prv[v].push_back(u);
        }
    }

    /*
        Count shortest prefixes/suffixes.
        We only use these numbers to choose a good meet layer.
    */
    vector<u64> waysL(n + 1, 0), waysR(n + 1, 0);

    waysL[1] = 1;

    for (int d = 0; d < D; ++d) {
        for (int u : layer[d]) {
            for (int v : nxt[u]) {
                waysL[v] += waysL[u];
            }
        }
    }

    waysR[n] = 1;

    for (int d = D; d > 0; --d) {
        for (int v : layer[d]) {
            for (int u : prv[v]) {
                waysR[u] += waysR[v];
            }
        }
    }

    // Pick a layer with the smallest total half-path enumeration.
    int mid = 0;
    u64 bestScore = numeric_limits<u64>::max();

    for (int d = 0; d <= D; ++d) {
        u64 score = 0;

        for (int v : layer[d])
            score += waysL[v] + waysR[v];

        if (score < bestScore) {
            bestScore = score;
            mid = d;
        }
    }

    /*
        Minimum possible toll of:
          1 -> v
          v -> n
        while staying in the shortest-path DAG.

        These let us prune half-paths that can never form a
        route with total toll <= C.
    */
    vector<u64> minL(n + 1, INF_COST);
    vector<u64> minR(n + 1, INF_COST);

    minL[1] = c[1];

    for (int d = 0; d < D; ++d) {
        for (int u : layer[d]) {
            for (int v : nxt[u]) {
                minL[v] =
                    min(minL[v], minL[u] + c[v]);
            }
        }
    }

    minR[n] = c[n];

    for (int d = D - 1; d >= 0; --d) {
        for (int u : layer[d]) {
            for (int v : nxt[u]) {
                minR[u] =
                    min(minR[u], c[u] + minR[v]);
            }
        }
    }

    // uint64_t addition naturally works modulo 2^64.
    u64 answer = 0;

    for (int meet : layer[mid]) {

        /*
            Cheapest complete shortest route through 'meet':

            minL[meet] + minR[meet] - c[meet]
        */
        if (minL[meet] + minR[meet] - c[meet] > C)
            continue;

        /*
            A prefix with toll x is potentially useful only if

              x + minR[meet] - c[meet] <= C.

            Same idea for suffixes.
        */
        u64 limitL = C - minR[meet] + c[meet];
        u64 limitR = C - minL[meet] + c[meet];

        vector<u64> left, right;

        // Enumerate all useful shortest prefixes 1 -> meet.
        auto genLeft =
            [&](auto&& self, int u, u64 sum) -> void {

            if (sum > limitL)
                return;

            if (u == 1) {
                left.push_back(sum);
                return;
            }

            for (int p : prv[u]) {
                self(self, p, sum + c[p]);
            }
        };

        // Enumerate all useful shortest suffixes meet -> n.
        auto genRight =
            [&](auto&& self, int u, u64 sum) -> void {

            if (sum > limitR)
                return;

            if (u == n) {
                right.push_back(sum);
                return;
            }

            for (int v : nxt[u]) {
                self(self, v, sum + c[v]);
            }
        };

        genLeft(genLeft, meet, c[meet]);
        genRight(genRight, meet, c[meet]);

        sort(left.begin(), left.end());
        sort(right.begin(), right.end());

        /*
            Full toll:
                a + b - c[meet] <= C

            equivalent to
                a + b <= C + c[meet].
        */
        u64 threshold = C + c[meet];

        size_t j = right.size();

        for (u64 a : left) {
            while (j > 0 &&
                   a + right[j - 1] > threshold) {
                --j;
            }

            answer += (u64)j;
        }
    }

    cout << answer << '\n';
    return 0;
}
```

For the sample, the two shortest paths have tolls \(7\) and \(8\), so the program outputs:

```text
2
```

The BFS and DAG construction are \(O(n+m)\). If \(L_v\) and \(R_v\) are the numbers of enumerated prefix/suffix paths for a meeting vertex \(v\), the meet-in-the-middle part costs roughly

$$
O\!\left(\sum_v
(L_v\log L_v+R_v\log R_v)\right),
$$

with memory \(O(\max_v(L_v+R_v))\). The \(n\le90\) constraint is what makes this exponential meet-in-the-middle enumeration practical.
